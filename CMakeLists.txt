cmake_minimum_required(VERSION 3.16)
project(ModeratedLMS VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(lms main.cpp)
target_include_directories(lms PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lms PRIVATE SQLite::SQLite3 OpenSSL::Crypto)
target_compile_options(lms PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(lms_rules_test tests/rules_test.cpp)
target_include_directories(lms_rules_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lms_rules_test PRIVATE OpenSSL::Crypto)
target_compile_options(lms_rules_test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME business_rules COMMAND lms_rules_test)
